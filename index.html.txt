<script>
const KEY='lestari_aviary_farm_pwa_v1';
const PHOTO_KEY='lestari_aviary_header_photo_v1';
const IDS=Array.from({length:60},(_,i)=>'N'+String(i+1).padStart(3,'0'));

function fresh(){
 return {version:5,boxes:IDS.map(id=>({id,n:0,eggs:0,note:''})),big:0,moves:[],deaths:[],hatches:[],unhatched:[],sales:[],adjustments:[]}
}

function load(){
 try{
  const d=JSON.parse(localStorage.getItem(KEY));
  if(d&&Array.isArray(d.boxes)&&d.boxes.length===60){
   d.boxes=d.boxes.map((b,i)=>({id:IDS[i],n:Math.max(0,Number(b.n)||0),eggs:Math.max(0,Number(b.eggs)||0),note:String(b.note||'')}));
   d.big=Math.max(0,Number(d.big)||0);
   d.moves=Array.isArray(d.moves)?d.moves:[];
   d.deaths=Array.isArray(d.deaths)?d.deaths:[];
   d.hatches=Array.isArray(d.hatches)?d.hatches:[];
   d.unhatched=Array.isArray(d.unhatched)?d.unhatched:[];
   d.sales=Array.isArray(d.sales)?d.sales:[];
   d.adjustments=Array.isArray(d.adjustments)?d.adjustments:[];
   d.version=5;
   return d;
  }
 }catch(e){}
 return fresh();
}

let data=load(),selected='N001',draftEgg=0,draftChick=0,hatchQty=0,unhatchedQty=0;
const $=id=>document.getElementById(id);
const box=id=>data.boxes.find(b=>b.id===id);
const totalEggs=()=>data.boxes.reduce((s,b)=>s+b.eggs,0);
const totalBoxChicks=()=>data.boxes.reduce((s,b)=>s+b.n,0);
const totalDead=()=>data.deaths.reduce((s,d)=>s+(Number(d.qty)||0),0);
const totalBigDead=()=>data.deaths.filter(d=>d.location==='sarang besar'||d.location==='big').reduce((s,d)=>s+(Number(d.qty)||0),0);
const totalUnhatched=()=>data.unhatched.reduce((s,d)=>s+(Number(d.qty)||0),0);
const totalSold=()=>data.sales.reduce((s,d)=>s+(Number(d.qty)||0),0);
const totalHatched=()=>data.hatches.reduce((s,d)=>s+(Number(d.qty)||0),0);
const activeBoxes=()=>data.boxes.filter(b=>b.eggs>0||b.n>0);
const stamp=()=>new Date().toLocaleString('ms-MY');

function save(){
 localStorage.setItem(KEY,JSON.stringify(data));
 render();
}

function safe(v){
 return String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

function render(){
 const eggs=totalEggs(),bc=totalBoxChicks(),big=data.big,live=bc+big,dead=totalDead();
 const active=activeBoxes().length,eggBoxes=data.boxes.filter(b=>b.eggs>0).length,chickBoxes=data.boxes.filter(b=>b.n>0).length,both=data.boxes.filter(b=>b.eggs>0&&b.n>0).length;
 const deathPct=live+dead?dead/(live+dead)*100:0;
 const unhatched=totalUnhatched(),hatched=totalHatched(),outcomes=unhatched+hatched,unhatchedPct=outcomes?unhatched/outcomes*100:0;

 $('sEggs').textContent=eggs;
 $('sChicks').textContent=bc;
 $('sBig').textContent=big;
 $('sActive').textContent=active+' / 60';

 $('aActive').textContent=active;
 $('aPercent').textContent=(active/60*100).toFixed(1)+'%';
 $('aEggAvg').textContent=eggBoxes?(eggs/eggBoxes).toFixed(2):'0.00';
 $('aChickAvg').textContent=chickBoxes?(bc/chickBoxes).toFixed(2):'0.00';
 $('aEggBoxes').textContent=eggBoxes;
 $('aChickBoxes').textContent=chickBoxes;
 $('aBoth').textContent=both;
 $('aLive').textContent=live;
 $('aDead').textContent=dead;
 $('aDeathPct').textContent=deathPct.toFixed(1)+'%';
 $('aDeathLabel').textContent=deathPct.toFixed(1)+'%';
 $('aDeathBar').style.width=Math.min(100,deathPct)+'%';
 $('aTotalLive').textContent=live;
 $('aTotalEggs').textContent=eggs;
 $('aUnhatched').textContent=unhatched;
 $('aUnhatchedPct').textContent=unhatchedPct.toFixed(1)+'%';
 $('aHatched').textContent=hatched;
 $('aOutcomes').textContent=outcomes;

 $('boxgrid').innerHTML=data.boxes.map(b=>`<button class="boxbtn ${b.eggs||b.n?'has':''} ${b.id===selected?'sel':''}" data-id="${b.id}"><strong>${b.id}</strong><span>${b.eggs}T · ${b.n}A</span></button>`).join('');
 document.querySelectorAll('[data-id]').forEach(el=>el.onclick=()=>selectBox(el.dataset.id));

 $('boxTitle').textContent='Kotak '+selected;
 $('eggNum').textContent=draftEgg;
 $('chickNum').textContent=draftChick;
 $('hatchNum').textContent=hatchQty;
 $('unhatchedNum').textContent=unhatchedQty;

 // PEMBETULAN: simpan pilihan kotak sebelum dropdown dibina semula.
 const fromSelect=$('fromBox');
 const savedFrom=fromSelect.value || 'N001';

 fromSelect.innerHTML=data.boxes.map(b=>
  `<option value="${b.id}">${b.id} (${b.n} anak)</option>`
 ).join('');

 // Pulihkan pilihan asal supaya tidak kembali ke N001 setiap kali render().
 fromSelect.value=box(savedFrom)?savedFrom:'N001';

 $('deathBox').innerHTML=data.boxes.map(b=>
  `<option value="${b.id}">${b.id} (${b.n} anak)</option>`
 ).join('');

 $('fromAvailable').textContent=
  'Anak tersedia dalam kotak: '+box(fromSelect.value).n;

 $('bigNum').textContent=big;
 $('bigDeathsTotal').textContent=totalBigDead();
 $('soldTotal').textContent=totalSold();
 $('saleAvailable').textContent='Anak tersedia dalam sarang besar: '+big;

 const events=[
  ...data.moves.map(x=>({...x,type:'Pindah'})),
  ...data.deaths.map(x=>({...x,type:'Kematian'})),
  ...data.hatches.map(x=>({...x,type:'Menetas'})),
  ...data.unhatched.map(x=>({...x,type:'Telur tak menetas'})),
  ...data.sales.map(x=>({...x,type:'Keluaran jualan'})),
  ...data.adjustments.map(x=>({...x,type:'Pembetulan'}))
 ].sort((a,b)=>(b.ts||0)-(a.ts||0));

 $('historyList').innerHTML=events.length?events.map(x=>
  `<div class="item"><strong>${safe(x.type)}: ${safe(x.description||((x.qty||'')+' '+(x.from||x.location||'')))}</strong><br><small>${safe(x.date||'')}</small></div>`
 ).join(''):'<p class="hint">Belum ada sejarah.</p>';

 $('unhatchedPlus').disabled=unhatchedQty>=box(selected).eggs;
 $('unhatchedMinus').disabled=unhatchedQty<=0;
}

function selectBox(id){
 selected=id;
 draftEgg=box(id).eggs;
 draftChick=box(id).n;
 hatchQty=0;
 unhatchedQty=0;
 $('note').value=box(id).note||'';
 render();
}

document.querySelectorAll('.tab').forEach(t=>t.onclick=()=>{
 document.querySelectorAll('.tab').forEach(x=>x.classList.toggle('active',x===t));
 document.querySelectorAll('.panel').forEach(x=>x.classList.toggle('active',x.id==='panel-'+t.dataset.tab));
});

$('eggMinus').onclick=()=>{draftEgg=Math.max(0,draftEgg-1);render()};
$('eggPlus').onclick=()=>{draftEgg=Math.min(999,draftEgg+1);render()};
$('chickMinus').onclick=()=>{draftChick=Math.max(0,draftChick-1);render()};
$('chickPlus').onclick=()=>{draftChick=Math.min(999,draftChick+1);render()};

$('saveBox').onclick=()=>{
 box(selected).eggs=draftEgg;
 box(selected).n=draftChick;
 box(selected).note=$('note').value.trim();
 hatchQty=Math.min(hatchQty,draftEgg);
 unhatchedQty=Math.min(unhatchedQty,draftEgg);
 save();
 alert('Rekod '+selected+' disimpan.');
};

$('hatchMinus').onclick=()=>{hatchQty=Math.max(0,hatchQty-1);render()};
$('hatchPlus').onclick=()=>{hatchQty=Math.min(box(selected).eggs,hatchQty+1);render()};

$('recordHatch').onclick=()=>{
 const b=box(selected);
 if(hatchQty<1)return alert('Pilih bilangan telur yang menetas.');
 if(hatchQty>b.eggs)return alert('Telur tidak mencukupi.');
 if(!confirm('Rekod '+hatchQty+' telur menetas dalam '+selected+'?'))return;
 b.eggs-=hatchQty;
 b.n+=hatchQty;
 data.hatches.push({date:stamp(),ts:Date.now(),qty:hatchQty,from:selected,description:hatchQty+' telur menetas dalam '+selected});
 draftEgg=b.eggs;
 draftChick=b.n;
 hatchQty=0;
 unhatchedQty=Math.min(unhatchedQty,b.eggs);
 save();
 alert('Rekod penetasan disimpan.');

function download(name,content,type){
 const blob=new Blob([content],{type});
 const url=URL.createObjectURL(blob);
 const a=document.createElement('a');
 a.href=url;
 a.download=name;
 document.body.appendChild(a);
 a.click();
 a.remove();
 setTimeout(()=>URL.revokeObjectURL(url),1000);
}

$('brandPhotoInput').onchange=async e=>{
 const file=e.target.files&&e.target.files[0];
 if(!file)return;
 if(!file.type.startsWith('image/')){alert('Sila pilih fail gambar.');e.target.value='';return}

 try{
  const src=await new Promise((resolve,reject)=>{
   const r=new FileReader();
   r.onload=()=>resolve(r.result);
   r.onerror=reject;
   r.readAsDataURL(file);
  });

  const img=new Image();
  img.src=src;
  await new Promise((resolve,reject)=>{img.onload=resolve;img.onerror=reject});

  const scale=Math.min(1,500/Math.max(img.width,img.height));
  const canvas=document.createElement('canvas');
  canvas.width=Math.max(1,Math.round(img.width*scale));
  canvas.height=Math.max(1,Math.round(img.height*scale));
  canvas.getContext('2d').drawImage(img,0,0,canvas.width,canvas.height);

  const compressed=canvas.toDataURL('image/jpeg',0.75);

  try{
   localStorage.setItem(PHOTO_KEY,compressed);
   showBrandPhoto(compressed);
   alert('Gambar header berjaya disimpan. Buat backup JSON untuk menyimpan salinan gambar.');
  }catch(storageError){
   alert('Storan pelayar penuh. Buat backup dan cuba gambar yang lebih kecil.');
  }
 }catch(err){
  alert('Gambar tidak dapat dibaca. Cuba pilih gambar lain.');
 }
 e.target.value='';
};
